\chapter*{Hilfsmittelverzeichnis}
\vspace{-5em}
\addcontentsline{toc}{chapter}{Hilfsmittelverzeichnis}
\begin{itemize}
    \item \textbf{ChatGPT (OpenAI):} KI-basierte Assistenz zur Unterstützung bei technischen, sprachlichen und konzeptionellen Fragestellungen. Ein Promptprotokoll befindet sich im Anhang.
    \item \textbf{Claude AI (Anthropic):} Ergänzende KI zur Ideengenerierung und Textüberarbeitung. Auch hierzu liegt ein Promptprotokoll im Anhang bei.
    \item \textbf{Draw.io und Miro:} Werkzeuge zur Erstellung von Diagrammen, Systemübersichten und visuellen Darstellungen.
    \item \textbf{LanguageTool:} Software zur Prüfung und Korrektur von Rechtschreibung und Grammatik.
    \item \textbf{Microsoft Excel:} Tabellenkalkulation zur Erstellung und Pflege des Projektzeitplans.
    \item \textbf{Cursor:} Editor auf Basis von Visual Studio Code mit KI-Unterstützung, verwendet zur Arbeit mit \LaTeX{}.
\end{itemize}
